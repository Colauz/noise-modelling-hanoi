% ============================================================================
%  Placeholder institutional marks.
%
%  These are NOT the institutions' logos. They are neutral typographic slugs
%  drawn so the deck's logo band is composed, aligned and consistently weighted
%  while the real files are being obtained. Each one is deliberately plain type
%  on a rule: it does not imitate any institution's actual mark, and it must not
%  be presented as one.
%
%  Replace them the moment you have the real files. Drop cosmos.pdf, vinuni.pdf,
%  isima.pdf and uca.pdf into this directory under exactly those names and the
%  deck picks them up with no edit to main.tex.
%
%  Rebuild:  make -C presentation logos-placeholder
% ============================================================================
\documentclass[border=1pt,multi=logoenv]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{xcolor}
\usepackage{tikz}

% The deck's own accent, so a placeholder never fights the slide it sits on.
\definecolor{accent}{HTML}{1F5C8B}
\definecolor{muted} {HTML}{6E6E6E}

\newenvironment{logoenv}{}{}

% \logomark{ACRONYM}{expansion} — every mark shares a cap height, a rule weight
% and a baseline, which is the whole point of drawing them rather than letting
% the band fall back to three differently-sized bits of running text.
\newcommand{\logomark}[2]{%
  \begin{logoenv}%
  \begin{tikzpicture}
    \node[anchor=base west, inner sep=0pt, text=accent] (a) at (0,0)
      {\fontfamily{lmss}\selectfont\bfseries\Large #1};
    \draw[accent, line width=.7pt]
      ([yshift=-1.7mm]a.base west) -- ([yshift=-1.7mm]a.base east);
    \node[anchor=north west, inner sep=0pt, text=muted]
      at ([yshift=-2.6mm]a.base west)
      {\fontfamily{lmss}\selectfont\scriptsize #2};
  \end{tikzpicture}%
  \end{logoenv}%
}

\begin{document}
\logomark{COSMOS}{Lab \textperiodcentered\ VinUniversity}
\logomark{VinUniversity}{Hanoi, Vietnam}
\logomark{ISIMA}{Clermont-Ferrand, France}
\logomark{UCA}{Universit\'e Clermont Auvergne}
\end{document}
