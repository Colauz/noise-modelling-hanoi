\begin{frame}{The gap this project sits in}

              out of reach for most local research budgets

    \end{column}

      \begin{alertblock}{The question}
        How far does a protocol built from \textbf{three consumer smartphones}
        and \textbf{open data} actually get you?
      \end{alertblock}
      \vspace{.5em}
      {\color{muted}The honest answer turned out to be: further than
      nothing, and much less far than a map.}
    \end{column}
  \end{columns}
\end{frame}

\section{Data and protocol}
% ============================================================================

\begin{frame}{The campaign}
  \fig[.48]{campaign}{data/processed/measurements.csv}
  \vspace{-.3em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Protocol} --- 3 handsets, \emph{Decibel X} (A-weighted, SLOW),

      $\rightarrow$ Kobo.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Published, CC-BY-4.0} --- both CSVs and both XLSForms.
      {\color{muted} Videos and raw Kobo exports are \textbf{not}: faces,
      plates, collector identities.}
    \end{column}
  \end{columns}
\end{frame}
\note{363 raw submissions, 363 published rows --- nothing was silently dropped, and
that is checkable from the raw export.

If asked about ethics: no ethics review was requested for the video collection and
none is claimed. Only non-identifying aggregates are released. The full statement is
docs/data-sources.md, and it says exactly that rather than dressing it up.}

% ----------------------------------------------------------------
\begin{frame}{Metrology --- the constraint everything else is built around}
  \begin{alertblock}{The three phones were cross-calibrated against \emph{each other}, never against a reference instrument}
    A bias common to all three is \textbf{invisible in the data by construction}.
  \end{alertblock}
  \vspace{.3em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{What that forbids}
      \begin{itemize}
        \item Stating an absolute level as a fact
        \item Any compliance claim under QCVN 26:2010 --- which requires a
              class 1 or class 2 instrument
        \item Comparing a 25\,s sample to an annual $L_\mathrm{den}$
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{What it still allows}
      \begin{itemize}
        \item Contrasts \emph{between places} --- bias cancels in a difference
        \item Contrasts \emph{between hours}
        \item A \emph{bounded} absolute-bias interval, from instrumented
              literature: \wn{$+3.0$ to $+7.3\dB$}
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  The target quantity is \LA{}: a 20--30\,s A-weighted sample. \textbf{Not}
  $L_\mathrm{Aeq}$, $L_\mathrm{den}$ or $L_\mathrm{night}$.
  \vspace{.2em}
  \src{docs/metrology.md \; $\cdot$ \; results/tables/literature\_anchoring.md} \; \aside{the interval is estimated, never applied}
\end{frame}
\note{This slide is the spine of the talk. Everything downstream --- the modest
\rr{}, the refusal to publish an exceedance rate, and ultimately the decision to
build our own app --- follows from one fact: nobody would lend us a class 1 meter,
and we could not buy one.

The +3.0 to +7.3 dB interval comes from anchoring our roadside-day stratum against
Phan 2010 and Gelb \& Apparicio 2019 after correcting for the quantity mismatch. It
bounds the bias; it does not correct it. No offset is written anywhere in the code
--- CALIBRATION\_OFFSET is 0.0 for all three collectors, deliberately.}

\begin{frame}{The evaluation protocol --- and why it is the whole story}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      Every model is scored under \textbf{three splits}, on exactly the same folds:
      \vspace{.4em}
      \begin{enumerate}
        \item \textbf{Block-CV, 600\,m} --- square blocks in UTM 48N,
              \texttt{GroupKFold} on the block. The permissive one.
        \item \textbf{Buffered leave-one-out, 300\,m} --- for each point, train on
              \emph{all} points more than 300\,m away.\\
              \up{The reference protocol.}
        \item \textbf{Leave-one-site-out} --- generalisation to an unseen urban
              typology.
      \end{enumerate}
      \vspace{.5em}
      Every score carries a 95\,\% interval bootstrapped \textbf{by block}, not by
      point: resampling correlated points gives an interval that is far too narrow.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{alertblock}{Why 300\,m and not less}
        The morphology features aggregate over a disc of \textbf{radius
        300\,m}. Two points 110\,m apart share more than \textbf{85\,\%} of that
        disc.
        \vspace{.3em}

        Group on 110\,m cells and the model sees near-twins of its test points in
        training. The score is then not out-of-sample.
      \end{alertblock}
      \aside{Roberts et al.\ 2017, \emph{Ecography}}\\ \src{scripts/04\_evaluate\_models.py}
    \end{column}
  \end{columns}
\end{frame}
\note{This is the methodological core. If the audience takes one slide away from
the method section, it should be this one.

The buffer must match the feature radius --- that is the rule, and it is the rule we
broke in July. Anyone doing spatial ML on aggregated features has probably broken
it too, which is why this slide usually starts a conversation.}

If someone asks why the hybrid wins block-CV: 600 m blocks still leave near-neighbours
in training when features span 300 m, so a flexible learner picks up local structure
it cannot generalise. That is exactly the effect Roberts 2017 describes.}

\begin{frame}{The delivered model}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      A line-source attenuation kernel. Two distances, three constants,
      \textbf{no morphology at all}:
      \begin{block}{}
        \centering
        $E = \dfrac{A_\mathrm{hw}}{\max(d_\mathrm{hw}, D_0)}
           + \dfrac{A_\mathrm{res}}{\max(d_\mathrm{res}, D_0)} + B$
        \vspace{.3em}

        $L = 10\log_{10} E$
      \end{block}
      \aside{$A_\mathrm{hw} = 4.774\times10^{7}$ \; $A_\mathrm{res} = 3.800\times10^{7}$ \; $D_0 = 5$\,m}
      \begin{itemize}
        \item Positivity-constrained --- every parameter is interpretable
        \item Ten lines of any language: no runtime, no server
        \item \wn{$B$ fits to $-99.6\dB$}, a boundary solution.
              \textbf{The background term is not identifiable here.}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Performance under the reference protocol}
        \begin{tabular}{@{}lr@{}}
          \rr{}       & \textbf{0.246} \\
          95\,\% CI   & $[0.097,\; 0.339]$ \\
          RMSE        & $6.20\dB$ \\
          MAE         & $5.01\dB$ \\
          \midrule
          data spread ($\sigma$) & $7.14\dB$ \\
          \bottomrule
        \end{tabular}
      \end{block}
      \begin{alertblock}{Say the error out loud}
        A typical prediction is wrong by \textbf{5\,dB} --- a factor of three
        in energy. Enough for a contrast, nowhere near an assessment.
      \end{alertblock}
    \end{column}
  \end{columns}
  \src{models/hybrid\_physical.json \; $\cdot$ \; models/metrics.json}
\end{frame}
\note{The degenerate B is ours to raise, not a reviewer's. Called a three-parameter
model, it is effectively two: the optimiser pushed the background to the boundary
because 363 points over three sites do not constrain a floor.

The 5 dB MAE is the number to repeat. Every time someone looks at the map and starts
planning something with it, that number is the answer.}

\begin{frame}{Three engineering results worth reporting}
  \begin{block}{GAMA cannot run on a phone --- and did not need to}
    GAMA is an Eclipse RCP desktop application: SWT, AWT, a full JVM,
    no Android port. But tier 1 is not an agent simulation --- it is a traffic
    multiplier over 5\,587 cells we already export. Reimplemented natively, it runs
    offline at 60\,fps. \textbf{The multiplier scales the traffic share of the
    energy and never the background} --- the correction that took the claimed
    benefit of pedestrianisation from $7.0\dB$ to $3.5\dB$.
  \end{block}
  \begin{block}{KoboToolbox renames your form at deployment}
    The first real submission failed with \textbf{404 for a form the server
    had never heard of}. Kobo discards the XLSForm's \texttt{id\_string} on deploy
    and substitutes the asset's own identifier
    ({\ttfamily\scriptsize aA8FaTuUVSkRjbUW7rCBz7}). The app now reads
    \texttt{/formList} and submits under whatever the server calls the form.
    \textbf{Nothing about this is visible from the phone} --- which is the argument
    for testing against the real server rather than reasoning about it.
  \end{block}
  \begin{block}{Consent is a version, not a flag}
    \texttt{Settings.CONSENT\_VERSION} is raised when what is collected
    changes, so consent is \textbf{asked again} rather than inherited from an
    agreement given to different facts. Declining leaves the map and the results
    usable --- they send nothing.
  \end{block}
\end{frame}
\note{All three are the kind of thing that costs a day and is invisible in a
results table, which is exactly why they are on a slide.

The 7.0 to 3.5 dB correction is a real scientific point hiding in an engineering
slide: scaling total energy by a traffic multiplier double-counts the background,
and it halved a mitigation benefit we had been about to report.}

\begin{frame}{What the app measures --- and what it refuses to claim}
  \begin{alertblock}{The app's level is a \emph{different quantity} from the campaign's}
    This handset's microphone, not a trimmed Decibel X reading. It goes in
    \textbf{its own field}, with the device model, the OS version and the audio
    source the platform granted. \textbf{The two never share a column.}
  \end{alertblock}
  \vspace{.3em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Verified end to end, against the live server}
      \begin{itemize}
        \item Fix, 25\,s measurement, instance written, \up{submission accepted
              by \texttt{kc.kobotoolbox.org}}
        \item The map draws all three areas; both sliders move it
        \item A position outside them is \textbf{refused, not answered}
        \item 36 unit tests; weighting pinned to the IEC table
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Not verified --- and it is the important one}
        \textbf{The level has never been checked against a sound source.} An
        emulator records silence.
        \vspace{.3em}

        Until it is compared to a meter, \textbf{the absolute figure means
        nothing} --- equally true, per \texttt{docs/metrology.md}, of the
        campaign's own numbers.
      \end{alertblock}
      \aside{First task on a real handset.}
    \end{column}
  \end{columns}
\end{frame}
\note{Do not oversell the app. It is a well-tested piece of software whose central
number is unvalidated, and saying that plainly is the same discipline as the rest of
the project.

The microphone offset in Settings is one number that stands for the whole of a
handset's absolute calibration. Its default is a plausible constant for a consumer
phone, not a measurement, and the app says so on the screen.

If asked about public distribution: an anonymous OpenRosa endpoint accepts whatever
is posted to it, and the free plan caps submissions. That is phase 6, and it is not
solved. Do not distribute an APK before it is.}

\section{Next: Clermont-Ferrand}
% ============================================================================

\begin{frame}{What Clermont-Ferrand unlocks that Hanoi could not}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{The binding constraint, restated}
        Every limitation in this study reduces to one fact: \textbf{no
        reference instrument was ever available}. Not borrowed, not rented, not
        bought.
      \end{alertblock}
      \vspace{.4em}
      \begin{exampleblock}{What changes}
        At ISIMA / Université Clermont Auvergne, a \textbf{class 1 or class 2
        sound level meter and a 94\,dB / 1\,kHz acoustic calibrator} are
        institutional equipment, not a budget line.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{The three upgrades, in dependency order}
      \vspace{.3em}
      \begin{enumerate}
        \item \up{Anchor the absolute scale.} Calibrate the app against the
              reference meter across levels and handsets. Turn
              \texttt{micOffset} from a plausible constant into an
              \textbf{affine, measured correction}. Everything downstream inherits
              this.
        \item \up{Validate the detector.} $\sim$10 videos, double-counted by hand,
              precision/recall/MAPE per class. \emph{One day of work that turns an
              open-ended limitation into a quantified uncertainty.}
        \item \up{Replace the kernel with real physics.} CNOSSOS-EU via
              \textbf{NoiseModelling}, corrected by a locally learned residual ---
              the architecture the data has been pointing at since August.
        \end{enumerate}
    \end{column}
  \end{columns}
  \vfill
  \src{docs/handover.md} \; \aside{\S6 ``the scientifically indicated next step'', debt \#2, P0-3}
\end{frame}
\note{Order matters and it is a dependency order, not a preference order. Nothing
about a propagation kernel is interesting while the levels it is fitted to are
uncalibrated --- a physically correct model fitted to a 5 dB offset is a physically
correct model of the wrong thing.

Upgrade 2 is the cheapest thing on the list and blocks the modal shares, which block
any emission model. One day.}

\section{Conclusion}
% ============================================================================

\begin{frame}{What this work established}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{exampleblock}{Findings}
        \begin{enumerate}
          \item At $n=363$ over three sites, \textbf{three physical parameters beat
                every learned model} we built. Capacity does not pay.
          \item \textbf{Cross-city transfer fails} --- \rr{} $<0$, with matched
                features.
          \item The video chain \textbf{could not be made to work}; that is an
                instrument result, not an acoustic one.
          \item A cross-calibrated smartphone protocol supports \textbf{contrasts},
                never levels.
        \end{enumerate}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Practices that made those findings trustworthy}
        \begin{itemize}
          \item The buffer matches the feature radius --- always
          \item Baselines and an ablation on \emph{the same} splits
          \item The delivered model is chosen \textbf{by code}
          \item Every number has \textbf{exactly one home}
          \item Retracted work is archived \textbf{with its reason}
          \item Nothing is predicted outside the sampled envelope
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    \small The deliverable is a \textbf{protocol, an instrument and an honest
    error bar} --- not a noise map of Hanoi.
  \end{center}
\end{frame}
\note{Close on the right-hand column if the room is technical. Those six practices
are transferable to any spatial ML project, and they are the part of this work most
likely to be reused by someone who never measures a decibel.

The last line is the thesis of the whole talk. Say it slowly.}
